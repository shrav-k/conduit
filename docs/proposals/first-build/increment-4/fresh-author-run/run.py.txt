import json, pathlib, subprocess, sys, datetime
BASE = pathlib.Path(__file__).resolve().parents[1]
BIN = str(BASE/'bin/conduit')
STORE = str(BASE/'store')
def log(s):
    with (BASE/'log.txt').open('a') as f:
        f.write(subprocess.check_output(['date','+%H:%M:%S'],text=True).strip()+' '+s+'\n')
def write(path, data):
    p=BASE/path
    p.parent.mkdir(parents=True,exist_ok=True)
    if p.exists(): raise RuntimeError('Refusing overwrite '+str(p))
    log('COMMAND: create '+str(path))
    p.write_text(json.dumps(data,indent=2)+'\n' if not isinstance(data,str) else data)
def run(label, args, mutate=True):
    if mutate: log('COMMAND '+label+': conduit '+' '.join(args))
    p=subprocess.run([BIN,args[0],'--store',STORE,*args[1:]],capture_output=True,text=True)
    write('project/evidence/'+label+'.jsonl',p.stdout)
    if p.stderr: write('project/evidence/'+label+'.stderr.txt',p.stderr)
    print(label,'exit',p.returncode,p.stdout,flush=True)
    return p
if __name__=='__main__':
    run(sys.argv[1],sys.argv[2:])
